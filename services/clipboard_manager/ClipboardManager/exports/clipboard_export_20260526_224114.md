# ============================================
# Touch Screen Toggle (Modular)
# ============================================

# --- Core Initialization ---
$CoreDir = Join-Path (Split-Path $PSScriptRoot -Parent) "core"
Import-Module (Join-Path $CoreDir "SystemCore.psm1")
Import-Module (Join-Path $CoreDir "TrayCore.psm1")
Hide-ConsoleWindow

# --- Security & Instance Lock ---
Request-AdminElevation -FilePath $MyInvocation.MyCommand.Path
$script:mutex = Get-SingleInstanceMutex -Name "Global\TouchToggleTray"
if ($null -eq $script:mutex) { exit }

# --- Config ---
$DEVICE_NAME = "HID-compliant touch screen"

# --- Helper: Get device & state ---
function Get-TouchStatus {
    $device = Get-PnpDevice | Where-Object { $_.FriendlyName -eq $DEVICE_NAME } | Select-Object -First 1
    return ($device -and $device.Status -eq "OK")
}

# --- Toggle Logic ---
function Toggle-TouchScreen {
    $device = Get-PnpDevice | Where-Object { $_.FriendlyName -eq $DEVICE_NAME } | Select-Object -First 1
    if ($device.Status -eq "OK") {
        Disable-PnpDevice -InstanceId $device.InstanceId -Confirm:$false
    } else {
        Enable-PnpDevice -InstanceId $device.InstanceId -Confirm:$false
    }
}

# --- Icon Logic ---
function Update-TrayState {
    $enabled = Get-TouchStatus
    $script:notifyIcon.Icon = New-CustomTrayIcon -DrawAction {
        param($g)
        $color = if ($enabled) { [System.Drawing.Color]::LimeGreen } else { [System.Drawing.Color]::Red }
        $brush = New-Object System.Drawing.SolidBrush($color)
        $g.FillEllipse($brush, 4, 4, 24, 24)
        
        # Draw "T" for Touch
        $font = New-Object System.Drawing.Font("Segoe UI", 10, [System.Drawing.FontStyle]::Bold)
        $g.DrawString("T", $font, [System.Drawing.Brushes]::White, 8, 6)
        $brush.Dispose(); $font.Dispose()
    }.GetNewClosure()
    $script:notifyIcon.Text = "Touch Screen: $(if($enabled){'Enabled'}else{'Disabled'})"
}

# --- Build Tray App or Headless Loop ---
if ($env:HUB_MANAGED -eq "1" -or $env:NO_ICON -eq "1") {
    while ($true) {
        Start-Sleep -Seconds 3600
    }
} else {
    $appContext = New-Object System.Windows.Forms.ApplicationContext
    $script:notifyIcon = New-Object System.Windows.Forms.NotifyIcon
    $script:notifyIcon.Visible = $true

    $contextMenu = New-Object System.Windows.Forms.ContextMenuStrip
    $contextMenu.Items.Add("Touch Toggle").Enabled = $false
    $contextMenu.Items.Add("Toggle State", $null, { Toggle-TouchScreen; Update-TrayState }) | Out-Null
    $contextMenu.Items.Add("-") | Out-Null
    $contextMenu.Items.Add("Exit", $null, { $script:notifyIcon.Visible = $false; $appContext.ExitThread() }) | Out-Null
    $script:notifyIcon.ContextMenuStrip = $contextMenu

    $script:notifyIcon.Add_MouseClick({
        param($sender, $e)
        if ($e.Button -eq [System.Windows.Forms.MouseButtons]::Left) { Toggle-TouchScreen; Update-TrayState }
    })

    Update-TrayState
    [System.Windows.Forms.Application]::Run($appContext)
}
